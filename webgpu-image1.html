// WebGPU requires bytesPerRow to be a multiple of 256 const bytesPerVoxel = 1; // 8-bit r8unorm const unpaddedBytesPerRow = ctVolume.dimX * bytesPerVoxel; // 128 const paddedBytesPerRow = Math.ceil(unpaddedBytesPerRow / 256) * 256; // 256 // Re-pack data with 256-byte row stride const paddedBuffer = new Uint8Array(paddedBytesPerRow * ctVolume.dimY * ctVolume.dimZ); for (let z = 0; z < ctVolume.dimZ; z++) { for (let y = 0; y < ctVolume.dimY; y++) { const srcIdx = (z * ctVolume.dimY + y) * ctVolume.dimX; const dstIdx = (z * ctVolume.dimY + y) * paddedBytesPerRow; paddedBuffer.set(ctVolume.data.subarray(srcIdx, srcIdx + ctVolume.dimX), dstIdx); } } // Upload padded buffer to VRAM device.queue.writeTexture( { texture: volumeTexture }, paddedBuffer, { bytesPerRow: paddedBytesPerRow, rowsPerImage: ctVolume.dimY }, [ctVolume.dimX, ctVolume.dimY, ctVolume.dimZ] );
